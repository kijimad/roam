* 概要

Go で定数インデックスで配列にアクセスすると、ビルド時にチェックされる。

#+begin_src sh
  set -e
  d=$(mktemp -d)
  cat > "$d/go.mod" <<'EOF'
  module demo

  go 1.26.0
  EOF
  cat > "$d/bce.go" <<'EOF'
  package demo

  func main() {
  var a [3]int
  _ = a[5]
  }
  EOF

  cd "$d"
  go build . 2>&1
#+end_src

#+RESULTS:
#+begin_src
./bce.go:5:7: invalid argument: index 5 out of bounds [0:3]
#+end_src
