* 概要

Go言語で、文字列を複数置換したいとき、Replacerを使うと簡単にできる。

#+begin_src go
  import (
          "fmt"
          "strings"
  )

  var dnaReplacer *strings.Replacer = strings.NewReplacer(
          "A", "T",
          "T", "A",
          "C", "G",
          "G", "C",
  )

  func DNAStrand(dna string) string {
          return dnaReplacer.Replace(dna)
  }

  func main() {
          fmt.Println(DNAStrand("ATCG"))
  }
#+end_src
