* 概要

OpenTelemetry の3つのシグナル(trace、metric、log)を、それぞれいちばん小さい例で出力して見る。どれも標準出力の exporter で JSON を書き出し、整形して出力をそのまま見る。

#+caption: 3つの例で使う共通部分
#+name: setup
#+begin_src shell :results silent
  # 例ごとに一時ディレクトリを作り、終わったら消す
  dir=$(mktemp -d)
  trap 'rm -rf "$dir"' EXIT
  cd "$dir"
  go mod init example >/dev/null 2>&1
  go get \
    go.opentelemetry.io/otel@v1.47.0 \
    go.opentelemetry.io/otel/sdk@v1.47.0 \
    go.opentelemetry.io/otel/sdk/metric@v1.47.0 \
    go.opentelemetry.io/otel/sdk/log@v1.47.0 \
    go.opentelemetry.io/otel/log@v1.47.0 \
    go.opentelemetry.io/otel/exporters/stdout/stdouttrace@v1.47.0 \
    go.opentelemetry.io/otel/exporters/stdout/stdoutmetric@v1.47.0 \
    go.opentelemetry.io/otel/exporters/stdout/stdoutlog@v0.23.0 2>/dev/null
#+end_src

** trace

親の span の中で子の span を作る。

#+caption: 親と子の span を1つずつ作る
#+begin_src shell :results output verbatim :noweb no-export
  <<setup>>
  mkdir -p trace
  cat > trace/main.go <<'GO'
  package main

  import (
  	"context"

  	"go.opentelemetry.io/otel/exporters/stdout/stdouttrace"
  	sdktrace "go.opentelemetry.io/otel/sdk/trace"
  )

  func main() {
  	ctx := context.Background()
  	exp, _ := stdouttrace.New(stdouttrace.WithPrettyPrint())
  	tp := sdktrace.NewTracerProvider(sdktrace.WithSyncer(exp))
  	defer tp.Shutdown(ctx)

  	tracer := tp.Tracer("example")
  	ctx, parent := tracer.Start(ctx, "parent")
  	_, child := tracer.Start(ctx, "child")
  	child.End()
  	parent.End()
  }
  GO
  go mod tidy 2>/dev/null
  go run ./trace
#+end_src

** metric

カウンタに 1、2、3 を足す。

#+caption: カウンタに3回足す
#+begin_src shell :results output verbatim :noweb no-export
  <<setup>>
  mkdir -p metric
  cat > metric/main.go <<'GO'
  package main

  import (
  	"context"

  	"go.opentelemetry.io/otel/exporters/stdout/stdoutmetric"
  	sdkmetric "go.opentelemetry.io/otel/sdk/metric"
  )

  func main() {
  	ctx := context.Background()
  	exp, _ := stdoutmetric.New(stdoutmetric.WithPrettyPrint())
  	mp := sdkmetric.NewMeterProvider(sdkmetric.WithReader(sdkmetric.NewPeriodicReader(exp)))
  	defer mp.Shutdown(ctx)

  	counter, _ := mp.Meter("example").Int64Counter("requests")
  	counter.Add(ctx, 1)
  	counter.Add(ctx, 2)
  	counter.Add(ctx, 3)
  }
  GO
  go mod tidy 2>/dev/null
  go run ./metric
#+end_src

** log

ログを1件書く。

#+caption: ログを1件書く
#+begin_src shell :results output verbatim :noweb no-export
  <<setup>>
  mkdir -p log
  cat > log/main.go <<'GO'
  package main

  import (
  	"context"

  	"go.opentelemetry.io/otel/attribute"
  	"go.opentelemetry.io/otel/exporters/stdout/stdoutlog"
  	"go.opentelemetry.io/otel/log"
  	sdklog "go.opentelemetry.io/otel/sdk/log"
  )

  func main() {
  	ctx := context.Background()
  	exp, _ := stdoutlog.New(stdoutlog.WithPrettyPrint())
  	lp := sdklog.NewLoggerProvider(sdklog.WithProcessor(sdklog.NewSimpleProcessor(exp)))
  	defer lp.Shutdown(ctx)

  	var r log.Record
  	r.SetSeverity(log.SeverityInfo)
  	r.SetBody(attribute.StringValue("hello"))
  	lp.Logger("example").Emit(ctx, r)
  }
  GO
  go mod tidy 2>/dev/null
  go run ./log
#+end_src
