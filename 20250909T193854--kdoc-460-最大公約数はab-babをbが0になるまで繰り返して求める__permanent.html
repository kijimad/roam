概要

数の概念が怪しい。パッと出てこないのでメモしておく。 

import "fmt"
func main() {
        fmt.Println(gcd(24, 4))
        fmt.Println(gcd(65536, 1248))
        fmt.Println(gcd(1248, 65536))
        fmt.Println(gcd(19, 2))
}

func gcd(a,b int)(int){
        for b!=0 {
                a,b = b,a%b
        }
        return a
}
