* 概要

Goの ~json.Unmarshal~ は、JSONに構造体のフィールドと対応しないキーがあっても、エラーにせず黙って捨てる。キーの綴りを間違えても気づけない。検知したいときは ~json.Decoder~ の ~DisallowUnknownFields~ を使う。

#+begin_src go
  package main

  import (
  	"bytes"
  	"encoding/json"
  	"fmt"
  )

  type User struct {
  	Name string
  }

  func main() {
  	data := []byte(`{"Name": "a", "Age": 20}`)

  	var u1 User
  	err := json.Unmarshal(data, &u1)
  	fmt.Printf("Unmarshal: %+v, err: %v\n", u1, err)

  	var u2 User
  	dec := json.NewDecoder(bytes.NewReader(data))
  	dec.DisallowUnknownFields()
  	err = dec.Decode(&u2)
  	fmt.Printf("DisallowUnknownFields: %+v, err: %v\n", u2, err)
  }
#+end_src

#+RESULTS:
#+begin_src
Unmarshal: {Name:a}, err: <nil>
DisallowUnknownFields: {Name:a}, err: json: unknown field "Age"
#+end_src

#+begin_src shell :results output
  go version
  go doc encoding/json | grep -A2 'unknown keys in the JSON object'
#+end_src
