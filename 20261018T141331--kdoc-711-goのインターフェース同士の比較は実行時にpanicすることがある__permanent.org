* 概要

Goのインターフェース同士の ~==~ は、コンパイルが通っても実行時にpanicすることがある。2つの値の動的型が同じで、その型が比較できない型(スライス、マップ、関数など)のときである。動的型が違えば、比較できない型でもpanicせずfalseになる。

#+begin_src go
  package main

  import "fmt"

  func eq(a, b any) (result string) {
  	defer func() {
  		if r := recover(); r != nil {
  			result = fmt.Sprint("panic: ", r)
  		}
  	}()
  	return fmt.Sprint(a == b)
  }

  func main() {
  	fmt.Println("int同士:", eq(1, 1))
  	fmt.Println("スライスとint:", eq([]int{1}, 1))
  	fmt.Println("スライス同士:", eq([]int{1}, []int{1}))
  }
#+end_src

#+RESULTS:
#+begin_src
int同士: true
スライスとint: false
スライス同士: panic: runtime error: comparing uncomparable type []int
#+end_src
